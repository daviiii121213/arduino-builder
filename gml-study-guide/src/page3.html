05Máquina de estados do jogadorIDLE⇄WALK → JUMP → FALL → IDLE · ATTACK → IDLE · HURT → IDLE/DEAD

          // Create
          estado = ESTADO.IDLE;   t_dano = 0;
          sprites = [spr_idle, spr_walk, spr_jump,
              spr_fall, spr_atk, spr_hurt, spr_dead];
          // Step
          var h = keyboard_check(vk_right)
                - keyboard_check(vk_left);
          var chao = place_meeting(x, y + 1, obj_bloco);
          var pulo = keyboard_check_pressed(vk_space);
          switch (estado) {
              case ESTADO.IDLE:
              case ESTADO.WALK:
                  hsp = h * 3;
                  estado = h != 0 ? ESTADO.WALK : ESTADO.IDLE;
        
              if (!chao) estado = ESTADO.FALL;
              if (pulo) { vsp = -7; estado = ESTADO.JUMP; }
              if (keyboard_check_pressed(ord("J"))) {
                  estado = ESTADO.ATTACK;
                  image_index = 0;
              }
              break;
          case ESTADO.JUMP:
              hsp = h * 3;
              if (vsp >= 0) estado = ESTADO.FALL;
              break;
          case ESTADO.FALL:
              hsp = h * 3;
              if (chao) estado = ESTADO.IDLE;   break;
        
              case ESTADO.ATTACK:
                  hsp = 0;
                  if (image_index >= image_number - 1)
                      estado = ESTADO.IDLE;
                  break;
              case ESTADO.HURT:  // t_dano = 20 ao ser atingido
                  t_dano -= 1;
                  if (t_dano <= 0) estado = ESTADO.IDLE;
                  if (hp <= 0)     estado = ESTADO.DEAD;
                  break;
              case ESTADO.DEAD:  hsp = 0;  break;
          }
          sprite_index = sprites[estado];  // enum = índice
          // + gravidade e colisão (Folha 2 §06)

            ini_open("config.ini");              // INI

            function salvar(_arq, _dados) {      // JSON
                var s = json_stringify(_dados);
                var b = buffer_create(1, buffer_grow, 1);
                buffer_write(b, buffer_string, s);
                buffer_save(b, _arq);   buffer_delete(b);
            }
            function carregar(_arq) {
                if (!file_exists(_arq)) return undefined;
                var b = buffer_load(_arq);
                var s = buffer_read(b, buffer_string);
                buffer_delete(b);   return json_parse(s);
            }

06Funções reutilizáveis

            /// @param {Id.Instance} _alvo  alvo do dano
            /// @return {Bool} true se o alvo morreu
            function aplicar_dano(_alvo, _dano, _forca = 2) {
                if (!instance_exists(_alvo)) return false;
                with (_alvo) {
                    hp -= _dano;
                    hsp = sign(x - other.x) * _forca;
                }
                return _alvo.hp <= 0;
            }

13Checklist de desempenho

  • Destrua o que sai da tela; evite criar instâncias em massa.
  • Sem laços aninhados sobre todas as instâncias no Step.
  • Colisão só quando precisa; máscaras simples.
  • Todo ds_*_create tem seu destroy.
  • Surfaces e partículas: crie 1×, reutilize, libere.
  • Arrays grandes: array_create(n); cache o tamanho.
  • Cálculos repetidos e temporários em var.
  • Objetos-pai + event_inherited(); meça com o Profiler.
