06Sistema de plataforma

            // Create
            hsp = 0;   vsp = 0;   grv = 0.4;
            vel = 3;   forca_pulo = -7;
            // Step — horizontal + gravidade
            hsp = (keyboard_check(vk_right)
                - keyboard_check(vk_left)) * vel;
            vsp = min(vsp + grv, 10);      // queda máx.
            var chao = place_meeting(x, y + 1, obj_bloco);
            if (chao && keyboard_check_pressed(vk_space))
                vsp = forca_pulo;           // pulo
            if (vsp < 0 && !keyboard_check(vk_space))
                vsp = max(vsp, forca_pulo / 2); // curto
            // parede: encosta 1 px por vez e para
            if (place_meeting(x + hsp, y, obj_bloco)) {
                var sx = sign(hsp);
                while (!place_meeting(x + sx, y, obj_bloco))
                    x += sx;
                hsp = 0;
            }
            x += hsp;
            if (place_meeting(x, y + vsp, obj_bloco)) { // y
                var sy = sign(vsp);
                while (!place_meeting(x, y + sy, obj_bloco))
                    y += sy;
                vsp = 0;
            }
            y += vsp;
DefiniçãoDireto: você soma a x/y (controle total, como acima). Por velocidade: define speed/hspeed e o GameMaker move sozinho — ideal para projéteis. Não misture.

11Sistemas do jogador

            // Create
            hp_max = 5;   hp = hp_max;   inv = 0;
            global.cp_x = x;   global.cp_y = y;
            tomar_dano = function(_dano) {
                if (inv > 0 || hp <= 0) return;
                hp = max(hp - _dano, 0);
                inv = 60;             // 1 s invencível
                if (hp == 0) {        // morte
                    visible = false;   alarm[1] = 90;
                }
            };
            // Step — pisca se invencível
            if (inv > 0) inv--;
            image_alpha = (inv mod 8 < 4) ? 1 : 0.3;
            // Alarm 1 — respawn no checkpoint
            x = global.cp_x;   y = global.cp_y;
            hp = hp_max; vsp = 0; inv = 90; visible = true;
            // obj_espinho · Collision com obj_player
            other.tomar_dano(1);
            // obj_checkpoint · Collision com obj_player
            global.cp_x = x;   global.cp_y = y;
